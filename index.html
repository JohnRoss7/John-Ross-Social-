async function postContent(){ const ta = document.getElementById('postText'); if(!ta.value.trim() && selectedFiles.length === 0) { return alert('Γράψε κάτι ή πρόσθεσε μια φωτογραφία!'); } const postBtn = document.getElementById('postSubmitBtn'); if(postBtn) { postBtn.disabled = true; postBtn.textContent = 'Αποστολή...'; } try { let imageUrls = []; for (let file of selectedFiles) { // Δημιουργούμε καθαρό όνομα αρχείου const fileName = `posts/${user.uid}/${Date.now()}_${Math.random().toString(36).substring(2, 9)}.jpg`; const storageRef = storage.ref(fileName); // Metadata για καλύτερη συμβατότητα const metadata = { contentType: file.type || 'image/jpeg' }; const snapshot = await storageRef.put(file, metadata); const downloadUrl = await snapshot.ref.getDownloadURL(); imageUrls.push(downloadUrl); } const docUser = await db.collection('users').doc(user.uid).get(); const u = docUser.data() || {}; await db.collection('posts').add({ uid: user.uid, name: u.name || user.displayName || user.email.split('@')[0], email: user.email, photo: u.photo || user.photoURL || '', text: ta.value || '', images: imageUrls, likes: 0, likedBy: [], comments: [], createdAt: Date.now() }); ta.value = ''; selectedFiles = []; updateImagePreview(); loadFeed(); } catch(err) { console.error(err); alert('Σφάλμα αποστολής:\n' + err.message); } finally { if(postBtn) { postBtn.disabled = false; postBtn.textContent = t('post'); } } }
