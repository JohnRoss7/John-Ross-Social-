async function likePost(id, el) { if (!user) { alert(lang === 'el' ? 'Πρέπει να συνδεθείς πρώτα' : 'Please sign in first'); return; } const ref = db.collection('posts').doc(id); const doc = await ref.get(); const p = doc.data(); const liked = (p.likedBy || []).includes(user.uid); if (liked) { await ref.update({ likes: firebase.firestore.FieldValue.increment(-1), likedBy: firebase.firestore.FieldValue.arrayRemove(user.uid) }); } else { await ref.update({ likes: firebase.firestore.FieldValue.increment(1), likedBy: firebase.firestore.FieldValue.arrayUnion(user.uid) }); playLikeSound(); if (el) { el.classList.add('like-pop'); setTimeout(() => el.classList.remove('like-pop'), 350); } if (p.uid !== user.uid) { const myName = await getUserName(user.uid); db.collection('notifications').add({ userId: p.uid, senderId: user.uid, postId: id, // σημαντικό! type: 'like', text: myName + (lang === 'el' ? ' έκανε like στη δημοσίευσή σου.' : ' liked your post.'), read: false, timestamp: Date.now() }); } } if (singlePostId) renderSinglePost(); else if (tab === 'profile') { setTimeout(() => { const feed = document.getElementById('myPostsFeed'); if (feed) loadProfilePosts(viewingUid || user.uid, feed); }, 280); } }
